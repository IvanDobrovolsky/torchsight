# Beam — training requirements
#
# Additional packages needed by scripts/train_lora.py and scripts/export_gguf.py.
# Requires a CUDA GPU. Beam v1.0 was trained on 8x A100 80GB SXM4 in 10.5 hours.
#
# Install with:
#   pip install -r requirements.txt
#   pip install -r requirements-training.txt
#
# The version pins matter: trl 0.11.4 + transformers 4.45.2 + peft 0.13.2 is
# the exact stack used for Beam v1.0. Newer versions of trl have moved
# SFTTrainer / SFTConfig fields around in ways that break the training script.

torch>=2.1
transformers==4.45.2
peft==0.13.2
trl==0.11.4
accelerate>=0.34
bitsandbytes>=0.43
sentencepiece>=0.2
deepspeed>=0.15   # optional: for multi-GPU training via train.sh

# GGUF export (export_gguf.py shells out to llama.cpp's convert script)
gguf>=0.10
